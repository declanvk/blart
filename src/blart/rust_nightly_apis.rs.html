4/// Writes a length prefix into this hasher, as part of being prefix-free.
5///
6/// If you're implementing [`Hash`] for a custom collection, call this before
7/// writing its contents to this `Hasher`.  That way
8/// `(collection![1, 2, 3], collection![4, 5])` and
9/// `(collection![1, 2], collection![3, 4, 5])` will provide different
10/// sequences of values to the `Hasher`
11///
12/// The `impl<T> Hash for [T]` includes a call to this method, so if you're
13/// hashing a slice (or array or vector) via its `Hash::hash` method,
14/// you should **not** call this yourself.
15///
16/// This method is only for providing domain separation.  If you want to
17/// hash a `usize` that represents part of the *data*, then it's important
18/// that you pass it to [`Hasher::write_usize`][core::hash::Hasher::write_usize]
19/// instead of to this method.
20///
21/// # Note to Implementers
22///
23/// If you've decided that your `Hasher` is willing to be susceptible to
24/// Hash-DoS attacks, then you might consider skipping hashing some or all
25/// of the `len` provided in the name of increased performance.
26///
27/// **This is a unstable API copied from the Rust standard library, tracking
28/// issue is [#96762][issue-96762]**
29///
30/// [issue-96762]: https://github.com/rust-lang/rust/issues/96762
31#[inline]
32pub fn hasher_write_length_prefix<H: core::hash::Hasher>(state: &mut H, num_entries: usize) {
33    cfg_select! {
34        feature = "nightly" => {
35            <H as core::hash::Hasher>::write_length_prefix(state, num_entries);
36        },
37        _ => {
38            state.write_usize(num_entries);
39        },
40    }
41}
