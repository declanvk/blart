19/// Split the tree rooted by `root` into two at the given `key_bytes`.
20///
21/// The keys less than `key_bytes` end up in one tree, and the keys greater than or equal to
22/// `key_bytes` end up in the other.
23///
24/// The split happens in three phases:
25///  1. Search for `key_bytes` from the root, recording every inner node on the path and which of
26///     its children are to the left and right of the path (see [`collect_split_point_nodes`]). The
27///     search stops at one of the following:
28///     - a leaf, which goes to the left if its key is less than `key_bytes` and to the right
29///       otherwise.
30///     - an inner node whose prefix does not match `key_bytes`. The whole node goes to the left or
31///       right depending on how its prefix compares to `key_bytes`.
32///     - an inner node with no child for the next key byte, which is split at that key byte.
33///  2. Cut the leaf linked list between the maximum leaf on the left and the minimum leaf on the
34///     right. If one side is empty, return the tree unchanged instead.
35///  3. Walk back up the path, splitting each inner node into a left and right node (see
36///     [`split_ancestor`]).
37///
38/// Leaf nodes are never split, each one is moved whole to one of the two trees. When an inner node
39/// is split, its allocation is reused for one side if the number of children on that side fits the
40///  node type, otherwise a new node of the right size is allocated. If a side has only one child,
41/// that level of the tree is collapsed into the child, by prepending the inner node's prefix and
42/// key byte to the child's prefix. Inner nodes that are not reused are deallocated.
43///
44/// # Safety
45///  - `root` must be the root of a well-formed tree, and every node of the tree must have been
46///    allocated with `alloc`.
47///  - This function requires exclusive access to the tree, no other operation may read or write any
48///    node of the tree for the duration of this function.
49///  - After this function returns, `root` must not be used except through the root pointers in the
50///    returned [`SplitOffResult`]. For a [`SplitOffResult::NormalSplit`], some of the inner nodes
51///    on the path to the split point have been modified or deallocated.
52///
53/// # Panics
54///
55/// This function calls [`AsBytes::as_bytes`] on keys in the tree, which may panic. Those calls all
56/// happen before the tree is modified, so a panic leaves the tree unchanged.
57pub unsafe fn split_off_unchecked<K: AsBytes, V, A: Allocator, const PREFIX_LEN: usize>(
58    root: OpaqueNodePtr<K, V, PREFIX_LEN>,
59    key_bytes: &[u8],
60    alloc: &A,
61) -> SplitOffResult<K, V, PREFIX_LEN> {
62    let CollectSplitPointResult { path, terminal } = unsafe {
63        // SAFETY: There is no concurrent mutation of the tree, covered by the safety requirements
64        // of this function
65        collect_split_point_nodes(root, key_bytes)
66    };
67
68    // Find the leaves on either side of the split before making any further changes, since we can
69    // figure out whether to return early based on these leaves.
70    //
71    // SAFETY: The tree has not been modified yet and there is no concurrent mutation, covered by
72    // the safety requirements of this function
73    let (max_left, min_right) = unsafe { find_split_boundary(&terminal) };
74    let (max_left, min_right) = match (max_left, min_right) {
75        (None, Some(_)) => {
76            return SplitOffResult::OnlyGreaterThanOrEqual {
77                greater_than_or_equal_root: root,
78            };
79        },
80        (Some(_), None) => {
81            return SplitOffResult::OnlyLessThan {
82                less_than_root: root,
83            };
84        },
85        (Some(max_left), Some(min_right)) => (max_left, min_right),
86        (None, None) => unreachable!("the tree contains at least one leaf"),
87    };
88
89    // Cut the linked list between the two trees
90    //
91    // SAFETY: We have exclusive access to the tree, covered by the safety requirements of this
92    // function. `max_left` and `min_right` are different leaves, and each mutable reference only
93    // lasts for its own statement.
94    unsafe {
95        max_left.as_mut().next = None;
96        min_right.as_mut().previous = None;
97    }
98
99    let (mut left, mut right) = match terminal {
100        CollectSplitPointTerminal::Leaf {
101            leaf_ptr,
102            ordering_to_split_key,
103        } => match ordering_to_split_key {
104            Ordering::Less => (Some(leaf_ptr.to_opaque()), None),
105            Ordering::Equal | Ordering::Greater => (None, Some(leaf_ptr.to_opaque())),
106        },
107        CollectSplitPointTerminal::InnerNodePrefixMismatch {
108            inner_node,
109            ordering_to_split_key,
110        } => match ordering_to_split_key {
111            Ordering::Less => (Some(inner_node.to_opaque()), None),
112            Ordering::Equal | Ordering::Greater => (None, Some(inner_node.to_opaque())),
113        },
114        CollectSplitPointTerminal::InnerNodeMissingChild(inner_node_path_elem) => {
115            // SAFETY: We have exclusive access to the tree and every node was allocated with
116            // `alloc`, covered by the safety requirements of this function. The element was just
117            // built by `collect_split_point_nodes` and the tree has not been restructured since
118            // (only the leaf linked list was cut), so it satisfies the `InnerNodePathElem`
119            // invariants. There is no child on the path, so no slot in the inner node is stale,
120            // and the children are exactly `left_children` plus `right_children`.
121            let InnerNodeSplit { left, right, pos } =
122                unsafe { inner_node_path_elem.split(key_bytes, None, alloc) };
123
124            // SAFETY: We have exclusive access to the tree. `split` puts each child of the inner
125            // node on exactly one side, and the inner node itself was either reused for a `Many`
126            // side or deallocated, so nothing else accesses a singleton child during this call.
127            // `pos` came from a prefix match against `key_bytes`, so its prefix range is in
128            // bounds.
129            unsafe {
130                (
131                    left.into_opaque(key_bytes, &pos),
132                    right.into_opaque(key_bytes, &pos),
133                )
134            }
135        },
136    };
137
138    for elem in path.into_iter().rev() {
139        // SAFETY: We have exclusive access to the tree and every node was allocated with `alloc`,
140        // covered by the safety requirements of this function. `elem` was built by
141        // `collect_split_point_nodes`, and its inner node has not been modified since, because
142        // the path is processed from the deepest element up and each split only touches its own
143        // inner node and the halves of its child. `left` and `right` are the halves of the path
144        // child of `elem`: on the first iteration they come from the terminal, and afterwards
145        // from splitting the element one level deeper, which is the path child of this one.
146        (left, right) = unsafe { split_ancestor(key_bytes, left, right, elem, alloc) };
147    }
148
149    match (left, right) {
150        (Some(less_than_root), Some(greater_than_or_equal_root)) => SplitOffResult::NormalSplit {
151            less_than_root,
152            less_than_root_max_leaf: max_left,
153            greater_than_or_equal_root,
154            greater_than_or_equal_min_leaf: min_right,
155        },
156        _ => unreachable!("both sides of the split must be non-empty"),
157    }
158}

160/// The result of [`split_off_unchecked`].
161pub enum SplitOffResult<K, V, const PREFIX_LEN: usize> {
162    /// The tree had keys on both sides of the split, and was split into two trees.
163    ///
164    /// The leaf linked list was cut between `less_than_root_max_leaf` and
165    /// `greater_than_or_equal_min_leaf`. The minimum leaf of the original tree is now the minimum
166    /// leaf of the less than tree, and the maximum leaf of the original tree is now the maximum
167    /// leaf of the greater than or equal tree.
168    NormalSplit {
169        /// The root of the tree with all keys less than the split key
170        less_than_root: OpaqueNodePtr<K, V, PREFIX_LEN>,
171        /// The maximum leaf of the less than tree
172        less_than_root_max_leaf: NodePtr<PREFIX_LEN, LeafNode<K, V, PREFIX_LEN>>,
173        /// The root of the tree with all keys greater than or equal to the split key
174        greater_than_or_equal_root: OpaqueNodePtr<K, V, PREFIX_LEN>,
175        /// The minimum leaf of the greater than or equal tree
176        greater_than_or_equal_min_leaf: NodePtr<PREFIX_LEN, LeafNode<K, V, PREFIX_LEN>>,
177    },
178    /// Every key in the tree is less than the split key, so the tree was not modified.
179    OnlyLessThan {
180        /// The root of the original tree
181        less_than_root: OpaqueNodePtr<K, V, PREFIX_LEN>,
182    },
183    /// Every key in the tree is greater than or equal to the split key, so the tree was not
184    /// modified.
185    OnlyGreaterThanOrEqual {
186        /// The root of the original tree
187        greater_than_or_equal_root: OpaqueNodePtr<K, V, PREFIX_LEN>,
188    },
189}

223/// Search for `key_bytes` from `root`, and collect the inner nodes on the path along with the node
224/// where the search stopped.
225///
226/// For each inner node on the path, the children are partitioned into those with a key byte less
227/// than the path's key byte (the left) and those with a greater key byte (the right). The child on
228/// the path is in neither.
229///
230/// This function does not modify the tree.
231///
232/// # Safety
233///  - This function cannot be called concurrently with any mutating operation on `root` or any
234///    child node of `root`.
235unsafe fn collect_split_point_nodes<K: AsBytes, V, const PREFIX_LEN: usize>(
236    root: OpaqueNodePtr<K, V, PREFIX_LEN>,
237    key_bytes: &[u8],
238) -> CollectSplitPointResult<K, V, PREFIX_LEN> {
239    /// Match the prefix of an inner node and look up the child for the next key byte. Returns the
240    /// child to continue the search with, or the terminal if the search stops at this node.
241    ///
242    /// On return, `*current_depth` is still less than or equal to `key_bytes.len()`. When this
243    /// returns [`ControlFlow::Continue`], `*current_depth` has advanced past the node's prefix and
244    /// the key byte of the returned child.
245    ///
246    /// # Safety
247    ///  - `inner_node_ptr` must point to a live inner node of type `N`, which is the root of a
248    ///    well-formed subtree (every node reachable from it is live).
249    ///  - This function cannot be called concurrently with any mutating operation on the node at
250    ///    `inner_node_ptr` or any of its descendants.
251    ///  - `*current_depth` must be less than or equal to `key_bytes.len()`.
252    unsafe fn visit_inner_node<K, V, N, const PREFIX_LEN: usize>(
253        inner_node_ptr: NodePtr<PREFIX_LEN, N>,
254        key_bytes: &[u8],
255        path: &mut Vec<InnerNodePathElem<K, V, PREFIX_LEN>>,
256        current_depth: &mut usize,
257    ) -> ControlFlow<CollectSplitPointTerminal<K, V, PREFIX_LEN>, OpaqueNodePtr<K, V, PREFIX_LEN>>
258    where
259        N: InnerNode<PREFIX_LEN, Key = K, Value = V>,
260        K: AsBytes,
261        NodePtr<PREFIX_LEN, N>: Into<ConcreteInnerNodePtr<K, V, PREFIX_LEN>>,
262    {
263        // SAFETY: `inner_node_ptr` points to a live node with no concurrent mutation, covered by
264        // the safety requirements of this function. The reference does not outlive this function.
265        let inner_node = unsafe { inner_node_ptr.as_ref() };
266
267        let depth_at_prefix_start = *current_depth;
268
269        // SAFETY: `depth_at_prefix_start <= key_bytes.len()`, covered by the safety requirements
270        // of this function.
271        match unsafe { inner_node.match_full_prefix(key_bytes, depth_at_prefix_start) } {
272            Ok(PrefixMatch { matched_bytes }) => {
273                // Since the prefix matched, advance the depth by the size of the prefix. On a
274                // match, `matched_bytes` counts pairs zipped from the prefix and
275                // `key_bytes[depth_at_prefix_start..]`, so the depth stays `<= key_bytes.len()`.
276                *current_depth += matched_bytes;
277
278                let next_key_fragment = if *current_depth < key_bytes.len() {
279                    key_bytes[*current_depth]
280                } else {
281                    // The key has insufficient bytes, so it is a strict prefix of every key under
282                    // this node. Shorter sequences sort first, so every key under this node is
283                    // greater than the split key. See the `split_off_key_ends_before_node` test.
284                    return ControlFlow::Break(
285                        CollectSplitPointTerminal::InnerNodePrefixMismatch {
286                            inner_node: inner_node_ptr.into(),
287                            ordering_to_split_key: Ordering::Greater,
288                        },
289                    );
290                };
291
292                let child_lookup = inner_node.lookup_child(next_key_fragment);
293                let left_children = inner_node.range(..next_key_fragment).collect();
294                // Exclude the child at `next_key_fragment` from both sides, since it lies on the
295                // path and will itself be split
296                let right_children = inner_node
297                    .range((Bound::Excluded(next_key_fragment), Bound::Unbounded))
298                    .collect();
299
300                let pos = InnerNodeKeyPosition {
301                    depth_at_prefix_start,
302                    prefix_length: matched_bytes,
303                    next_key_fragment,
304                };
305
306                match child_lookup {
307                    Some(child_ptr) => {
308                        // Since the prefix matched and it found a child, advance the depth by 1
309                        // more key byte. This branch is only reached when
310                        // `*current_depth < key_bytes.len()`, so the depth stays
311                        // `<= key_bytes.len()`.
312                        *current_depth += 1;
313
314                        path.push(InnerNodePathElem {
315                            inner_node_ptr: inner_node_ptr.into(),
316                            left_children,
317                            right_children,
318                            pos,
319                        });
320                        ControlFlow::Continue(child_ptr)
321                    },
322                    None => ControlFlow::Break(CollectSplitPointTerminal::InnerNodeMissingChild(
323                        InnerNodePathElem {
324                            inner_node_ptr: inner_node_ptr.into(),
325                            left_children,
326                            right_children,
327                            pos,
328                        },
329                    )),
330                }
331            },
332            Err(mismatch) => {
333                ControlFlow::Break(CollectSplitPointTerminal::InnerNodePrefixMismatch {
334                    inner_node: inner_node_ptr.into(),
335                    ordering_to_split_key: mismatch.node_prefix_comparison_to_search_key_segment,
336                })
337            },
338        }
339    }
340
341    let mut current_node = root;
342    let mut path = Vec::new();
343    let mut current_depth = 0;
344
345    loop {
346        let control = match_concrete_node_ptr!(match (current_node.to_node_ptr()) {
347            InnerNode(inner_node_ptr) => {
348                // SAFETY: `current_node` is either `root` or a child returned by a previous
349                // visit, so it is a live node of the tree, and there is no concurrent mutation,
350                // covered by the safety requirements of this function. `current_depth` starts at
351                // 0, and each previous visit kept it `<= key_bytes.len()`.
352                unsafe {
353                    visit_inner_node(inner_node_ptr, key_bytes, &mut path, &mut current_depth)
354                }
355            },
356            LeafNode(leaf_ptr) => {
357                // SAFETY: There is no concurrent mutation of the tree, covered by the safety
358                // requirements of this function. The reference does not outlive this block.
359                let leaf_key = unsafe { leaf_ptr.as_key_ref() };
360                // This call may panic, since it is user-provided, we just need to ensure that there
361                // are no memory leaks and we don't leave the tree in an invalid state
362                let leaf_key_bytes = leaf_key.as_bytes();
363                let ordering_to_split_key = leaf_key_bytes.cmp(key_bytes);
364
365                return CollectSplitPointResult {
366                    path,
367                    terminal: CollectSplitPointTerminal::Leaf {
368                        leaf_ptr,
369                        ordering_to_split_key,
370                    },
371                };
372            },
373        });
374
375        current_node = match control {
376            ControlFlow::Continue(next_node) => next_node,
377            ControlFlow::Break(terminal) => return CollectSplitPointResult { path, terminal },
378        };
379    }
380}

400/// The node where the search for the split key stopped, and how it relates to the split key.
401enum CollectSplitPointTerminal<K, V, const PREFIX_LEN: usize> {
402    /// The search reached a leaf
403    Leaf {
404        leaf_ptr: NodePtr<PREFIX_LEN, LeafNode<K, V, PREFIX_LEN>>,
405        /// The ordering of the leaf's key compared to the split key
406        ordering_to_split_key: Ordering,
407    },
408    /// The prefix of an inner node did not match the split key.
409    ///
410    /// This also covers the insufficient bytes case, since we could view it as a "mismatch" of
411    /// length. In that case the `ordering_to_split_key` would always be `Ordering::Greater`
412    /// because the split key is a strict prefix of every key in the node, and shorter
413    /// sequences are always ordered before longer sequences in a lexicographic ordering.
414    InnerNodePrefixMismatch {
415        inner_node: ConcreteInnerNodePtr<K, V, PREFIX_LEN>,
416        /// The ordering of every key under the inner node compared to the split key
417        ordering_to_split_key: Ordering,
418    },
419    /// The prefix of an inner node matched, but the child lookup for the next key byte failed
420    InnerNodeMissingChild(InnerNodePathElem<K, V, PREFIX_LEN>),
421}
422
423impl<K, V, const PREFIX_LEN: usize> fmt::Debug for CollectSplitPointTerminal<K, V, PREFIX_LEN> {
424    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
425        match self {
426            Self::Leaf {
427                leaf_ptr,
428                ordering_to_split_key,
429            } => f
430                .debug_struct("Leaf")
431                .field("leaf_ptr", leaf_ptr)
432                .field("ordering_to_split_key", ordering_to_split_key)
433                .finish(),
434            Self::InnerNodePrefixMismatch {
435                inner_node,
436                ordering_to_split_key,
437            } => f
438                .debug_struct("InnerNodePrefixMismatch")
439                .field("inner_node", inner_node)
440                .field("ordering_to_split_key", ordering_to_split_key)
441                .finish(),
442            Self::InnerNodeMissingChild(elem) => {
443                f.debug_tuple("InnerNodeMissingChild").field(elem).finish()
444            },
445        }
446    }
447}

449/// Find the pair of adjacent leaves where the split happens, returning the maximum leaf less than
450/// the split key and the minimum leaf greater than or equal to the split key.
451///
452/// Either leaf is `None` when there are no leaves on that side of the split.
453///
454/// # Safety
455///  - This function cannot be called concurrently with any mutating operation on the tree that
456///    `terminal` was collected from.
457unsafe fn find_split_boundary<K, V, const PREFIX_LEN: usize>(
458    terminal: &CollectSplitPointTerminal<K, V, PREFIX_LEN>,
459) -> SplitLeafNodes<K, V, PREFIX_LEN> {
460    // SAFETY: Covered by the safety requirements of the containing function
461    let from_max_left = |max_left: NodePtr<PREFIX_LEN, LeafNode<K, V, PREFIX_LEN>>| {
462        (Some(max_left), unsafe { max_left.as_ref() }.next)
463    };
464    // SAFETY: Covered by the safety requirements of the containing function
465    let from_min_right = |min_right: NodePtr<PREFIX_LEN, LeafNode<K, V, PREFIX_LEN>>| {
466        (unsafe { min_right.as_ref() }.previous, Some(min_right))
467    };
468
469    match terminal {
470        CollectSplitPointTerminal::Leaf {
471            leaf_ptr,
472            ordering_to_split_key,
473        } => match ordering_to_split_key {
474            Ordering::Less => from_max_left(*leaf_ptr),
475            Ordering::Equal | Ordering::Greater => from_min_right(*leaf_ptr),
476        },
477        CollectSplitPointTerminal::InnerNodePrefixMismatch {
478            inner_node,
479            ordering_to_split_key,
480        } => match ordering_to_split_key {
481            // SAFETY: Covered by the safety requirements of the containing function
482            Ordering::Less => from_max_left(unsafe { maximum_unchecked(inner_node.to_opaque()) }),
483            Ordering::Equal | Ordering::Greater => {
484                // SAFETY: Covered by the safety requirements of the containing function
485                from_min_right(unsafe { minimum_unchecked(inner_node.to_opaque()) })
486            },
487        },
488        CollectSplitPointTerminal::InnerNodeMissingChild(elem) => {
489            if let Some((_, child)) = elem.left_children.last() {
490                // SAFETY: Covered by the safety requirements of the containing function
491                from_max_left(unsafe { maximum_unchecked(*child) })
492            } else if let Some((_, child)) = elem.right_children.first() {
493                // SAFETY: Covered by the safety requirements of the containing function
494                from_min_right(unsafe { minimum_unchecked(*child) })
495            } else {
496                unreachable!("inner node must have at least one child")
497            }
498        },
499    }
500}

507/// Split an inner node on the path, given the two halves of its child on the path.
508///
509/// `left_child` and `right_child` are the result of splitting the child on the path (the child at
510/// `elem.pos.next_key_fragment`), and either may be `None` if that side is empty. Returns the left
511/// and right halves of the inner node in `elem`, in the same form.
512///
513/// This works because every key under the path child has the same key byte at this inner node,
514/// so both halves of the path child keep that key byte. Every left sibling has a smaller key byte,
515/// so the left half goes after them. Every right sibling has a larger key byte, so the right half
516/// goes before them. After that, the inner node is split the same way as one with no child on the
517/// path, except that its slot for the path child is stale (see [`InnerNodePathElem::split`]).
518///
519/// # Safety
520///  - The caller must have exclusive access to the inner node of `elem` and every node reachable
521///    from it, and no other operation may read or write those nodes for the duration of this
522///    function.
523///  - Every node reachable from the inner node of `elem`, and every node in the subtrees of
524///    `left_child` and `right_child`, must have been allocated with `alloc`.
525///  - `elem` must satisfy the [`InnerNodePathElem`] invariants, and its inner node must have a
526///    child at `elem.pos.next_key_fragment` (the path child). That is, `elem` was pushed onto the
527///    path by [`collect_split_point_nodes`], and its inner node has not been modified since.
528///  - `left_child` and `right_child` must be the result of splitting the path child. Each must
529///    point to a live node. They must be different nodes, and neither may be in
530///    `elem.left_children` or `elem.right_children`. Apart from the stale path slot in the inner
531///    node of `elem`, no node outside their own subtrees may point to them or to any node in their
532///    subtrees.
533///  - `key_bytes` must be the same key bytes that `elem` was collected with.
534///
535/// After this function returns, the inner node of `elem` must not be used except through the
536/// returned pointers, since it was either reused for one side or deallocated.
537unsafe fn split_ancestor<K: AsBytes, V, const PREFIX_LEN: usize>(
538    key_bytes: &[u8],
539    left_child: Option<OpaqueNodePtr<K, V, PREFIX_LEN>>,
540    right_child: Option<OpaqueNodePtr<K, V, PREFIX_LEN>>,
541    mut elem: InnerNodePathElem<K, V, PREFIX_LEN>,
542    alloc: &impl Allocator,
543) -> SplitNodes<K, V, PREFIX_LEN> {
544    let path_key_fragment = elem.pos.next_key_fragment;
545    if let Some(left_child) = left_child {
546        // Every left sibling has a smaller key byte, so this sorts last
547        elem.left_children.push((path_key_fragment, left_child));
548    }
549    if let Some(right_child) = right_child {
550        // Every right sibling has a larger key byte, so this sorts first
551        // TODO(opt): avoid shifting the whole vector
552        elem.right_children
553            .insert(0, (path_key_fragment, right_child));
554    }
555
556    // SAFETY: Exclusive access and the allocator are covered by the safety requirements of this
557    // function. `elem` satisfied the `InnerNodePathElem` invariants on entry, and adding at most
558    // one half of the path child to each side at the path key byte preserves them: every left
559    // sibling has a smaller key byte and every right sibling a larger one, so both vectors stay
560    // strictly ascending. The inner node has a child at `path_key_fragment`, which is the stale
561    // slot. The halves are live, not already in either vector, and not pointed to by any other
562    // node except through that stale slot, covered by the safety requirements of this function.
563    let InnerNodeSplit { left, right, pos } =
564        unsafe { elem.split(key_bytes, Some(path_key_fragment), alloc) };
565
566    // SAFETY: Exclusive access is covered by the safety requirements of this function. `split`
567    // puts each child of the inner node, and each half of the path child, on exactly one side. The
568    // inner node was either reused for a `Many` side or deallocated, so nothing else accesses a
569    // singleton child during this call. `pos` came from a prefix match against `key_bytes`, so its
570    // prefix range is in bounds.
571    unsafe {
572        (
573            left.into_opaque(key_bytes, &pos),
574            right.into_opaque(key_bytes, &pos),
575        )
576    }
577}
578
579type SplitNodes<K, V, const PREFIX_LEN: usize> = (
580    Option<OpaqueNodePtr<K, V, PREFIX_LEN>>,
581    Option<OpaqueNodePtr<K, V, PREFIX_LEN>>,
582);
583
584/// An inner node that the split key passed through, with its children partitioned around the
585/// split key.
586///
587/// # Invariants
588///
589/// Let `node` be the inner node at `inner_node_ptr` and `b` be `pos.next_key_fragment`. The type
590/// does not enforce these. [`collect_split_point_nodes`] establishes them, [`split_ancestor`]
591/// preserves them, and [`InnerNodePathElem::split`] requires them:
592///  - **Partition:** for each child of `node` at key byte `k`, if `k < b` then `(k, child)` appears
593///    exactly once in `left_children`, and if `k > b` then it appears exactly once in
594///    `right_children`. The only other entries allowed are at most one entry with key byte `b` in
595///    each vector, which are halves of the path child added by [`split_ancestor`]. So every child
596///    of `node` other than the one at `b` ends up on exactly one side, which is what makes it sound
597///    to give each side ownership of its children.
598///  - **Ordering:** `left_children` and `right_children` are each strictly ascending by key byte.
599///  - **Prefix:** for the key bytes that this element was collected with,
600///    `pos.depth_at_prefix_start + pos.prefix_length < key_bytes.len()`. The full prefix of `node`
601///    equals `key_bytes[pos.depth_at_prefix_start..][..pos.prefix_length]`.
602///  - `node` is live and has not been modified since the element was collected.
603struct InnerNodePathElem<K, V, const PREFIX_LEN: usize> {
604    /// Pointer to the inner node
605    inner_node_ptr: ConcreteInnerNodePtr<K, V, PREFIX_LEN>,
606    /// The children with a key byte less than `pos.next_key_fragment`, in order
607    left_children: Vec<(u8, OpaqueNodePtr<K, V, PREFIX_LEN>)>,
608    /// The children with a key byte greater than `pos.next_key_fragment`, in order
609    right_children: Vec<(u8, OpaqueNodePtr<K, V, PREFIX_LEN>)>,
610    /// The position of the inner node relative to the split key
611    pos: InnerNodeKeyPosition,
612}
613
614impl<K, V, const PREFIX_LEN: usize> fmt::Debug for InnerNodePathElem<K, V, PREFIX_LEN> {
615    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
616        f.debug_struct("InnerNodePathElem")
617            .field("inner_node_ptr", &self.inner_node_ptr)
618            .field("left_children", &self.left_children)
619            .field("right_children", &self.right_children)
620            .field("pos", &self.pos)
621            .finish()
622    }
623}
624
625impl<K, V, const PREFIX_LEN: usize> InnerNodePathElem<K, V, PREFIX_LEN> {
626    /// Split the inner node into a left side made from `left_children` and a right side made from
627    /// `right_children`.
628    ///
629    /// Each side either reuses the original inner node, becomes a single child, or is a newly
630    /// allocated inner node. Only one side can reuse the original node, and if neither does, it is
631    /// deallocated.
632    ///
633    /// `stale_path_slot` is the key byte of the child on the path, if there is one. The original
634    /// inner node still has a slot at that key byte pointing to the child as it was before it was
635    /// split. That pointer is stale, the child may have been reused for one of its halves or
636    /// deallocated. The halves of the path child (if any) are already in `left_children` and
637    /// `right_children`, so the stale slot must not be copied over if the original node is reused.
638    ///
639    /// # Safety
640    ///  - The caller must have exclusive access to the inner node at `self.inner_node_ptr` and
641    ///    every node in `left_children` and `right_children`. No other operation may read or write
642    ///    them for the duration of this function.
643    ///  - The inner node must have been allocated with `alloc`.
644    ///  - `self` must satisfy the [`InnerNodePathElem`] invariants, and `key_bytes` must be the
645    ///    same key bytes that `self` was collected with.
646    ///  - If `stale_path_slot` is `None`, `left_children` and `right_children` must have no entries
647    ///    at `self.pos.next_key_fragment`, and the inner node must have no child there. If
648    ///    `stale_path_slot` is `Some(b)`, then `b` must equal `self.pos.next_key_fragment`, and the
649    ///    inner node must have a child at `b`.
650    ///  - Every node in `left_children` and `right_children` must be live. The only node allowed to
651    ///    point to them is the inner node, through its own slots or the stale path slot.
652    ///
653    /// After this function returns, `self.inner_node_ptr` must not be used except through the
654    /// returned [`SplitSide::Many`], since the inner node was either reused for one side or
655    /// deallocated. Each entry of `left_children` and `right_children` ends up on exactly one side.
656    unsafe fn split(
657        self,
658        key_bytes: &[u8],
659        stale_path_slot: Option<u8>,
660        alloc: &impl Allocator,
661    ) -> InnerNodeSplit<K, V, PREFIX_LEN> {
662        /// Allocate the new inner node at the smallest size that will fit the given list of
663        /// children, then insert those children.
664        fn allocate_node_for_children<K, V, const PREFIX_LEN: usize>(
665            key_bytes: &[u8],
666            children: &Vec<(u8, OpaqueNodePtr<K, V, PREFIX_LEN>)>,
667            pos: &InnerNodeKeyPosition,
668            alloc: &impl Allocator,
669        ) -> ConcreteInnerNodePtr<K, V, PREFIX_LEN> {
670            let [(first_byte, first_child), (second_byte, second_child), ..] = children.as_slice()
671            else {
672                unreachable!(
673                    "children array should have at least two elements, had instead {} elements",
674                    children.len()
675                )
676            };
677
678            let children_rest = &children.as_slice()[2..];
679
680            let prefix_bytes = &key_bytes
681                [pos.depth_at_prefix_start..(pos.depth_at_prefix_start + pos.prefix_length)];
682
683            macro_rules! build_inner_node {
684                ($t:ident) => {{
685                    let mut builder = $t::builder(prefix_bytes, pos.prefix_length)
686                        .write_child(*first_byte, *first_child)
687                        .write_child(*second_byte, *second_child);
688
689                    for (byte, child) in children_rest {
690                        builder = builder.write_child(*byte, *child);
691                    }
692
693                    NodePtr::allocate_node_ptr(builder.build(), alloc).into()
694                }};
695            }
696
697            match InnerNodeType::inner_node_type_for(children.len()) {
698                InnerNodeType::Node4 => {
699                    build_inner_node!(InnerNode4)
700                },
701                InnerNodeType::Node16 => {
702                    build_inner_node!(InnerNode16)
703                },
704                InnerNodeType::Node48 => {
705                    build_inner_node!(InnerNode48)
706                },
707                InnerNodeType::Node256 => {
708                    build_inner_node!(InnerNodeDirect)
709                },
710            }
711        }
712
713        /// Build one side of the split from `keep`, either by reusing the original inner node,
714        /// collapsing to a single child, or allocating a new inner node.
715        ///
716        /// If the original inner node is reused, `*did_consume_parent` is set to `true`, and the
717        /// node's children become exactly the entries of `keep`.
718        ///
719        /// # Safety
720        ///  - If `*did_consume_parent` is `false`, `inner_node_ptr` must point to a live inner node
721        ///    of type `N` that has not been reused for the other side. The caller must have
722        ///    exclusive access to it, and no other operation may read or write it for the duration
723        ///    of this function.
724        ///  - `keep` and `discard` must be the two sides of an [`InnerNodePathElem`] that satisfies
725        ///    its invariants for this inner node and `key_bytes`, and `stale_path_slot` must
726        ///    satisfy the requirements of [`InnerNodePathElem::split`].
727        ///  - Every node in `keep` must be live. The only node allowed to point to it is the inner
728        ///    node, through its own slots or the stale path slot.
729        #[expect(clippy::too_many_arguments)]
730        unsafe fn split_side<K, V, const PREFIX_LEN: usize, N>(
731            inner_node_ptr: NodePtr<PREFIX_LEN, N>,
732            key_bytes: &[u8],
733            keep: &Vec<(u8, OpaqueNodePtr<K, V, PREFIX_LEN>)>,
734            discard: &[(u8, OpaqueNodePtr<K, V, PREFIX_LEN>)],
735            stale_path_slot: Option<u8>,
736            pos: &InnerNodeKeyPosition,
737            did_consume_parent: &mut bool,
738            alloc: &impl Allocator,
739        ) -> SplitSide<K, V, PREFIX_LEN>
740        where
741            N: InnerNode<PREFIX_LEN, Key = K, Value = V>,
742            NodePtr<PREFIX_LEN, N>: Into<ConcreteInnerNodePtr<K, V, PREFIX_LEN>>,
743        {
744            if keep.is_empty() {
745                SplitSide::Empty
746            } else if !*did_consume_parent && N::TYPE.capacity_range().contains(&keep.len()) {
747                *did_consume_parent = true;
748
749                // we can re-use the original inner node and just remove the other side's children
750
751                // SAFETY: `*did_consume_parent` was `false`, so `inner_node_ptr` points to a live
752                // node that we have exclusive access to, covered by the safety requirements of this
753                // function. The reference does not outlive this block.
754                let inner_node = unsafe { inner_node_ptr.as_mut() };
755
756                // The slot for the path child points to the child before it was split, so it is
757                // stale no matter which side this is. Remove it first, so that it is gone even if
758                // this side has no half of the path child, and so the loop below only has to
759                // remove siblings.
760                if let Some(stale_key_fragment) = stale_path_slot {
761                    let removed = inner_node.remove_child(stale_key_fragment);
762                    debug_assert!(removed.is_some());
763                }
764
765                // TODO(opt): optimize removal of many children
766                for (key_fragment, _) in discard {
767                    if Some(*key_fragment) == stale_path_slot {
768                        // Already removed above
769                        continue;
770                    }
771
772                    let removed = inner_node.remove_child(*key_fragment);
773                    debug_assert!(removed.is_some());
774                }
775
776                // If this side has a half of the path child, write it back under the same key byte
777                // in place of the stale pointer.
778                if let Some(stale_key_fragment) = stale_path_slot
779                    && let Some((_, child)) =
780                        keep.iter().find(|(key, _)| *key == stale_key_fragment)
781                {
782                    inner_node.write_child(stale_key_fragment, *child);
783                }
784
785                SplitSide::Many {
786                    finished: inner_node_ptr.into(),
787                }
788            } else if let [(key_byte, node_ptr)] = keep.as_slice() {
789                SplitSide::Singleton {
790                    key_byte: *key_byte,
791                    node_ptr: *node_ptr,
792                }
793            } else {
794                // we need to allocate a new inner node for these children, since the number of
795                // children does not fit in the existing inner node.
796
797                let new_inner_node = allocate_node_for_children(key_bytes, keep, pos, alloc);
798
799                SplitSide::Many {
800                    finished: new_inner_node,
801                }
802            }
803        }
804
805        /// Split the inner node at `inner_node_ptr`, see [`InnerNodePathElem::split`].
806        ///
807        /// # Safety
808        ///  - Same as [`InnerNodePathElem::split`], where `inner_node_ptr`, `left_children`,
809        ///    `right_children`, and `pos` are the fields of the [`InnerNodePathElem`], and
810        ///    `inner_node_ptr` points to an inner node of type `N`.
811        unsafe fn split_inner_node<K, V, const PREFIX_LEN: usize, N>(
812            inner_node_ptr: NodePtr<PREFIX_LEN, N>,
813            key_bytes: &[u8],
814            left_children: Vec<(u8, OpaqueNodePtr<K, V, PREFIX_LEN>)>,
815            right_children: Vec<(u8, OpaqueNodePtr<K, V, PREFIX_LEN>)>,
816            stale_path_slot: Option<u8>,
817            pos: InnerNodeKeyPosition,
818            alloc: &impl Allocator,
819        ) -> InnerNodeSplit<K, V, PREFIX_LEN>
820        where
821            N: InnerNode<PREFIX_LEN, Key = K, Value = V>,
822            NodePtr<PREFIX_LEN, N>: Into<ConcreteInnerNodePtr<K, V, PREFIX_LEN>>,
823        {
824            let mut did_consume_parent = false;
825            // SAFETY: `did_consume_parent` is `false`, and the inner node is live and exclusively
826            // accessed, covered by the safety requirements of this function. `left_children` and
827            // `right_children` are the two sides of the element, and the left side's nodes are
828            // live and only pointed to by the inner node.
829            let left = unsafe {
830                split_side(
831                    inner_node_ptr,
832                    key_bytes,
833                    &left_children,
834                    &right_children,
835                    stale_path_slot,
836                    &pos,
837                    &mut did_consume_parent,
838                    alloc,
839                )
840            };
841            // SAFETY: If `did_consume_parent` is still `false`, the left side did not touch the
842            // inner node (it built an `Empty`, a `Singleton`, or a new node from
843            // `left_children`), so the inner node is still live and exclusively accessed. If it is
844            // `true`, this call does not access the inner node. Either way, the left side never
845            // accesses the nodes in `right_children`. The remaining requirements are the same as
846            // for the left side.
847            let right = unsafe {
848                split_side(
849                    inner_node_ptr,
850                    key_bytes,
851                    &right_children,
852                    &left_children,
853                    stale_path_slot,
854                    &pos,
855                    &mut did_consume_parent,
856                    alloc,
857                )
858            };
859
860            if !did_consume_parent {
861                // If the original inner node was not used, then we need to deallocate it
862
863                // SAFETY: Neither side reused the original inner node. By the partition invariant
864                // of `InnerNodePathElem`, every child other than the stale path slot was moved to
865                // one of the sides. So no returned side points to the inner node, and the caller
866                // must not use `inner_node_ptr` again, which means it is only deallocated once.
867                // The inner node was allocated with `alloc`, covered by the safety requirements of
868                // this function.
869                let _ = unsafe { NodePtr::deallocate_node_ptr(inner_node_ptr, alloc) };
870            }
871
872            InnerNodeSplit { left, right, pos }
873        }
874
875        match_concrete_inner_node_ptr!(match (self.inner_node_ptr) {
876            // SAFETY: Covered by the safety requirements of this function, the arguments are the
877            // fields of `self`, and `inner_node_ptr` has the concrete type of the inner node.
878            InnerNode(inner_node_ptr) => unsafe {
879                split_inner_node(
880                    inner_node_ptr,
881                    key_bytes,
882                    self.left_children,
883                    self.right_children,
884                    stale_path_slot,
885                    self.pos,
886                    alloc,
887                )
888            },
889        })
890    }
891}

893/// The result of [`InnerNodePathElem::split`].
894struct InnerNodeSplit<K, V, const PREFIX_LEN: usize> {
895    left: SplitSide<K, V, PREFIX_LEN>,
896    right: SplitSide<K, V, PREFIX_LEN>,
897    pos: InnerNodeKeyPosition,
898}
899
900impl<K, V, const PREFIX_LEN: usize> fmt::Debug for InnerNodeSplit<K, V, PREFIX_LEN> {
901    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
902        f.debug_struct("InnerNodeSplit")
903            .field("left", &self.left)
904            .field("right", &self.right)
905            .field("pos", &self.pos)
906            .finish()
907    }
908}

910/// One side of a split inner node.
911enum SplitSide<K, V, const PREFIX_LEN: usize> {
912    /// The side has no children
913    Empty,
914    /// The side has a single child, so the inner node level will be collapsed into it
915    Singleton {
916        key_byte: u8,
917        node_ptr: OpaqueNodePtr<K, V, PREFIX_LEN>,
918    },
919    /// The side has enough children to need an inner node, either the reused original or a new one
920    Many {
921        finished: ConcreteInnerNodePtr<K, V, PREFIX_LEN>,
922    },
923}
924
925impl<K, V, const PREFIX_LEN: usize> fmt::Debug for SplitSide<K, V, PREFIX_LEN> {
926    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
927        match self {
928            Self::Empty => f.write_str("Empty"),
929            Self::Singleton { key_byte, node_ptr } => f
930                .debug_struct("Singleton")
931                .field("key_byte", key_byte)
932                .field("node_ptr", node_ptr)
933                .finish(),
934            Self::Many { finished } => f.debug_struct("Many").field("finished", finished).finish(),
935        }
936    }
937}

939/// This struct represents the position of an inner node with respect to the split key.
940#[derive(Debug, Copy, Clone)]
941struct InnerNodeKeyPosition {
942    /// Position in split key matching the start of the inner node prefix
943    depth_at_prefix_start: usize,
944    /// The length of the inner node prefix
945    prefix_length: usize,
946    /// The key byte corresponding to the next child in the split path
947    next_key_fragment: u8,
948}
949
950impl<K, V, const PREFIX_LEN: usize> SplitSide<K, V, PREFIX_LEN> {
951    /// Convert this split result into the node pointer that should replace the original inner
952    /// node, collapsing a [`Singleton`][Self::Singleton] into its parent's position.
953    ///
954    /// For a [`Singleton`][Self::Singleton] child that is an inner node, its prefix is rewritten
955    /// to be the parent's prefix, then `key_byte`, then its old prefix. The returned pointer takes
956    /// the place of the original inner node, and the caller is responsible for linking it into
957    /// exactly one parent (or using it as a root).
958    ///
959    /// # Safety
960    ///  - For a [`Singleton`][Self::Singleton], `node_ptr` must point to a live node. For the
961    ///    duration of this function, the node's memory must not be accessed (read or written)
962    ///    through any other pointer, as required by [`OpaqueNodePtr::header_mut`].
963    ///  - `pos` must be the position of the original inner node, from a prefix match against
964    ///    `key_bytes`. So `pos.depth_at_prefix_start + pos.prefix_length <= key_bytes.len()`, and
965    ///    that range of `key_bytes` is the full prefix of the original inner node.
966    unsafe fn into_opaque(
967        self,
968        key_bytes: &[u8],
969        pos: &InnerNodeKeyPosition,
970    ) -> Option<OpaqueNodePtr<K, V, PREFIX_LEN>> {
971        match self {
972            SplitSide::Empty => None,
973            SplitSide::Singleton { key_byte, node_ptr } => {
974                // SAFETY: `node_ptr` is live and not accessed through any other pointer during
975                // this function, covered by the safety requirements of this function. The
976                // reference only lasts for the scope of this `if` block.
977                if let Some(child_header) = unsafe { node_ptr.header_mut() } {
978                    // The removed node's prefix matched the search key, so its full prefix (even
979                    // past `PREFIX_LEN`) can be read from `key_bytes`.
980                    let parent_prefix = &key_bytes[pos.depth_at_prefix_start
981                        ..(pos.depth_at_prefix_start + pos.prefix_length)];
982
983                    let (old_prefix, old_len, old_capped_len) = child_header.clear_prefix();
984                    child_header.push_prefix(parent_prefix, pos.prefix_length);
985                    child_header.push_prefix(&[key_byte], 1);
986                    child_header.push_prefix(&old_prefix[..old_capped_len], old_len);
987                }
988                // the else case here is that the child does not have a header, and is a leaf
989
990                Some(node_ptr)
991            },
992            SplitSide::Many { finished } => Some(finished.to_opaque()),
993        }
994    }
995}
