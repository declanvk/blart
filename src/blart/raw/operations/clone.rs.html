15/// The result of cloning a trie
16#[derive(Debug)]
17pub struct CloneResult<K, V, const PREFIX_LEN: usize> {
18    /// The new root node of the trie
19    pub root: OpaqueNodePtr<K, V, PREFIX_LEN>,
20    /// The first leaf in the trie, aka the leaf with the minimum key
21    pub min_leaf: NodePtr<PREFIX_LEN, LeafNode<K, V, PREFIX_LEN>>,
22    /// The last leaf in the trie, aka the leaf with the maximum key
23    pub max_leaf: NodePtr<PREFIX_LEN, LeafNode<K, V, PREFIX_LEN>>,
24}

26/// Drop guard that deallocates a partially-cloned trie if a user-provided
27/// `Clone` implementation (for `K`, `V`, or a node header) panics partway
28/// through [`clone_unchecked`].
29struct PartialCloneGuard<'a, K, V, A: Allocator, const PREFIX_LEN: usize> {
30    /// Root of the partially-built new trie, or `None` if no node has been
31    /// allocated yet.
32    new_root_node: Option<OpaqueNodePtr<K, V, PREFIX_LEN>>,
33    /// The allocator that every node of the new trie was allocated with.
34    alloc: &'a A,
35}
36
37impl<K, V, A: Allocator, const PREFIX_LEN: usize> PartialCloneGuard<'_, K, V, A, PREFIX_LEN> {
38    fn into_root(mut self) -> Option<OpaqueNodePtr<K, V, PREFIX_LEN>> {
39        self.new_root_node.take()
40    }
41}
42
43impl<K, V, A: Allocator, const PREFIX_LEN: usize> Drop
44    for PartialCloneGuard<'_, K, V, A, PREFIX_LEN>
45{
46    fn drop(&mut self) {
47        if let Some(root) = self.new_root_node {
48            // SAFETY:
49            //  - This guard is the sole owner of the partially-built new trie, so this is the only
50            //    call to `deallocate_tree` for these nodes; there is no double-free.
51            //  - The new trie is not shared with any other thread and no references into it are
52            //    live at this point, so there is no concurrent read and no use-after-free.
53            //  - `alloc` is the same allocator that every node of the new trie was allocated with.
54            //  - Every child pointer written into the partial trie points to a fully allocated
55            //    node, so the traversal only ever visits valid nodes.
56            unsafe {
57                let _ = deallocate_tree(root, self.alloc);
58            }
59        }
60    }
61}

63/// Clone the given trie and all the key-values paired contained.
64///
65/// This function does not use recursion to clone, so it should not cause stack
66/// overflow when cloning a deep trie.
67///
68/// # Safety
69///
70/// This function can only be called while there are no mutable references to
71/// any of the nodes in this trie.
72pub unsafe fn clone_unchecked<
73    K: Clone + AsBytes,
74    V: Clone,
75    A: Allocator,
76    const PREFIX_LEN: usize,
77>(
78    root: OpaqueNodePtr<K, V, PREFIX_LEN>,
79    alloc: &A,
80) -> CloneResult<K, V, PREFIX_LEN> {
81    /// Update the current parent node with the newly cloned child, inserting it
82    /// at the given key byte.
83    ///
84    /// If the parent is finished, meaning it has the expected number of
85    /// children, this function pops that node off the unfinished stack.
86    ///
87    /// This function also updates the "new root node", which is inferred by
88    /// recording the node that is passed when the unfinished stack is empty.
89    ///
90    /// # Safety
91    ///
92    /// This function can only be called while there are no other mutable
93    /// references to the nodes of the cloned (in-progress) trie
94    unsafe fn update_parent<N, K, V, const PREFIX_LEN: usize>(
95        unfinished_nodes_stack: &mut Vec<(usize, ConcreteInnerNodePtr<K, V, PREFIX_LEN>)>,
96        new_root_node: &mut Option<OpaqueNodePtr<K, V, PREFIX_LEN>>,
97        key_byte: u8,
98        new_node_ptr: NodePtr<PREFIX_LEN, N>,
99    ) where
100        N: Node<PREFIX_LEN, Key = K, Value = V>,
101    {
102        let Some((expected_num_children, parent)) = unfinished_nodes_stack.last() else {
103            debug_assert!(
104                new_root_node.is_none(),
105                "First cloned node [{new_root_node:?}] is present while unfinished node stack is \
106                 empty",
107            );
108            // We know this is the root node since it has no parent
109            *new_root_node = Some(new_node_ptr.to_opaque());
110            return;
111        };
112
113        let updated_num_children = match_concrete_inner_node_ptr!(match (parent) {
114            InnerNode(inner_ptr) => {
115                // SAFETY: Covered by function safety doc AND the lifetime of this mutable
116                // reference is scoped to this block, and does not escape
117                let inner = unsafe { inner_ptr.as_mut() };
118                inner.write_child(key_byte, new_node_ptr.to_opaque());
119                inner.header().num_children()
120            },
121        });
122
123        debug_assert!(
124            updated_num_children <= *expected_num_children,
125            "should never add more [{updated_num_children}] than the expected number of children \
126             [{expected_num_children}]"
127        );
128        if updated_num_children == *expected_num_children {
129            // This inner node is finished, all the children have been added
130            let _ = unfinished_nodes_stack.pop().expect("last was Some");
131        }
132    }
133
134    /// This function starts cloning the inner node, and updates the various
135    /// stacks so that the new node will be updated as tree traversal continues.
136    ///
137    /// # Safety
138    ///
139    ///  - This function can only be called while there are no mutable references to any of the
140    ///    nodes in the trie referenced by `inner_ptr`.
141    ///  - Additionally, there must not be any other mutable references to nodes in the
142    ///    `unfinished_nodes_stack` stack.
143    unsafe fn clone_inner_node<N, K, V, A, const PREFIX_LEN: usize>(
144        unfinished_nodes_stack: &mut Vec<(usize, ConcreteInnerNodePtr<K, V, PREFIX_LEN>)>,
145        dfs_stack: &mut Vec<(u8, OpaqueNodePtr<K, V, PREFIX_LEN>)>,
146        new_root_node: &mut Option<OpaqueNodePtr<K, V, PREFIX_LEN>>,
147        key_byte: u8,
148        inner_ptr: NodePtr<PREFIX_LEN, N>,
149        alloc: &A,
150    ) where
151        N: InnerNode<PREFIX_LEN, Key = K, Value = V>,
152        ConcreteInnerNodePtr<K, V, PREFIX_LEN>: From<NodePtr<PREFIX_LEN, N>>,
153        A: Allocator,
154    {
155        // SAFETY: Covered by function safety comment
156        let inner = unsafe { inner_ptr.as_ref() };
157
158        // We add the node children in reverse order so that the first child lands on
159        // the top of the stack. That ensures we iterate DFS and visit the nodes
160        // in-order, not reversed
161        dfs_stack.extend(inner.iter().rev());
162
163        let mut header = inner.header().clone();
164        header.reset_num_children();
165        let new_node = N::from_header(header);
166        // At this point the header records 0 children, but has a copy of the prefix
167        // from the original node There are no child pointers in the node
168
169        let new_node_ptr = NodePtr::allocate_node_ptr(new_node, alloc);
170
171        // SAFETY: Covered by function safety doc, since there are no other mutable
172        // references to nodes in the `unfinished_nodes_stack` stack.
173        unsafe {
174            update_parent(
175                unfinished_nodes_stack,
176                new_root_node,
177                key_byte,
178                new_node_ptr,
179            )
180        };
181
182        // Get the expected number of children from the original header
183        let expected_num_children = inner.header().num_children();
184        unfinished_nodes_stack.push((
185            expected_num_children,
186            ConcreteInnerNodePtr::from(new_node_ptr),
187        ));
188    }
189
190    // (expected number of children, Pointer to inner node)
191    let mut unfinished_nodes_stack: Vec<(usize, ConcreteInnerNodePtr<K, V, PREFIX_LEN>)> =
192        Vec::new();
193    // (Key byte of this node, node pointer)
194    let mut dfs_stack: Vec<(u8, OpaqueNodePtr<K, V, PREFIX_LEN>)> = Vec::new();
195
196    // Guard deallocates partial root on unwinding
197    let mut guard = PartialCloneGuard {
198        new_root_node: None,
199        alloc,
200    };
201    let mut first_leaf: Option<NodePtr<PREFIX_LEN, LeafNode<K, V, PREFIX_LEN>>> = None;
202    let mut previous_leaf: Option<NodePtr<PREFIX_LEN, LeafNode<K, V, PREFIX_LEN>>> = None;
203
204    // We're okay to fabricate a key_byte for the root, since it will never be
205    // inserted into a parent node
206    dfs_stack.push((u8::MAX, root));
207
208    while let Some((key_byte, current_ptr)) = dfs_stack.pop() {
209        match_concrete_node_ptr!(match (current_ptr.to_node_ptr()) {
210            InnerNode(inner_ptr) => unsafe {
211                // SAFETY: The `clone_unchecked` safety comment covers other mutable references
212                // to the original trie, and we guarantee here there are no mutable references
213                // to nodes in the `unfinished_node_stack`
214                clone_inner_node(
215                    &mut unfinished_nodes_stack,
216                    &mut dfs_stack,
217                    &mut guard.new_root_node,
218                    key_byte,
219                    inner_ptr,
220                    alloc,
221                );
222            },
223            LeafNode(leaf_ptr) => {
224                // SAFETY: The `clone_unchecked` safety comment covers other mutable references
225                // to the original trie
226                let leaf = unsafe { leaf_ptr.as_ref() };
227
228                let new_leaf = leaf.clone_without_siblings();
229                let new_leaf_ptr = NodePtr::allocate_node_ptr(new_leaf, alloc);
230                if let Some(previous_ptr) = previous_leaf {
231                    // SAFETY: The `clone_unchecked` safety comment covers other mutable references
232                    // to the original trie
233                    unsafe { LeafNode::insert_after(new_leaf_ptr, previous_ptr) };
234                }
235                previous_leaf = Some(new_leaf_ptr);
236                let _ = first_leaf.get_or_insert(new_leaf_ptr);
237
238                // SAFETY: There are no mutable references to any nodes in the
239                // `unfinished_nodes_stack`
240                unsafe {
241                    update_parent(
242                        &mut unfinished_nodes_stack,
243                        &mut guard.new_root_node,
244                        key_byte,
245                        new_leaf_ptr,
246                    )
247                };
248            },
249        });
250    }
251
252    assert!(
253        unfinished_nodes_stack.is_empty(),
254        "The list of unfinished nodes must be empty when exiting the loop"
255    );
256
257    let new_root_node = guard.into_root();
258
259    CloneResult {
260        root: new_root_node.expect("there should be at least one cloned node"),
261        min_leaf: first_leaf.expect("should visit at least one leaf in a non-empty trie"),
262        max_leaf: previous_leaf.expect("should visit at least one leaf in a non-empty trie"),
263    }
264}

298    }

332    #[test]
333    fn clone_panicking_value_does_not_leak() {
334        use std::{
335            panic::{catch_unwind, AssertUnwindSafe},
336            sync::atomic::{AtomicUsize, Ordering},
337        };
338
339        static REMAINING_CLONES: AtomicUsize = AtomicUsize::new(0);
340
341        #[derive(PartialEq, Eq, Debug)]
342        struct PanicAfter(u32);
343
344        impl Clone for PanicAfter {
345            fn clone(&self) -> Self {
346                let left = REMAINING_CLONES.fetch_sub(1, Ordering::Relaxed);
347                assert!(left > 0, "boom: PanicAfter cloned too many times");
348                PanicAfter(self.0)
349            }
350        }
351
352        let mut tree: TreeMap<[u8; 2], PanicAfter> = TreeMap::new();
353        tree.try_insert([0, 0], PanicAfter(0)).unwrap();
354        for i in 1..=16u8 {
355            tree.try_insert([1, i], PanicAfter(i.into())).unwrap();
356        }
357
358        // At this point the tree should be something like:
359        // ```
360        //         node4
361        //         |   |
362        //         0   node16
363        //             | .. |
364        //             1 - 16
365        // ```
366        //
367        // Setting the number of remaining clones to 4 means that the first leaf is cloned and we
368        // have two levels of partially constructed leaf nodes before the unwinding starts. I think
369        // this should be sufficiently complicated and will exercise the deallocation with two
370        // partial inner nodes of different kinds.
371        REMAINING_CLONES.store(4, Ordering::Relaxed);
372
373        let result = catch_unwind(AssertUnwindSafe(|| {
374            let _ = tree.clone();
375        }));
376        assert!(
377            result.is_err(),
378            "clone should have unwound when the value `Clone` panicked"
379        );
380
381        assert_eq!(tree.get(&[0, 0]), Some(&PanicAfter(0)));
382        for i in 1..=16u8 {
383            assert_eq!(tree.get(&[1, i]), Some(&PanicAfter(i.into())));
384        }
385    }
